CREATE OR REPLACE FUNCTION public.bump_daily_activity() RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE
  v_course INTEGER;
  prev_pos INT := 0;
  new_pos  INT := 0;
  delta    INT := 0;
BEGIN
  SELECT course_id INTO v_course FROM course_assets WHERE id = NEW.asset_id;
  prev_pos := LEAST(COALESCE(OLD.position_sec,0), COALESCE(NEW.duration_sec, OLD.duration_sec, 0));
  new_pos  := LEAST(COALESCE(NEW.position_sec,0), COALESCE(NEW.duration_sec, OLD.duration_sec, 0));
  delta    := GREATEST(new_pos - prev_pos, 0);
  IF delta > 0 THEN
    INSERT INTO daily_learning_activity (user_id, course_id, activity_date, seconds_watched, assets_viewed)
    VALUES (NEW.user_id, v_course, CURRENT_DATE, delta, 0)
    ON CONFLICT (user_id, course_id, activity_date)
    DO UPDATE SET seconds_watched = daily_learning_activity.seconds_watched + EXCLUDED.seconds_watched;
  END IF;
  IF TG_OP = 'INSERT' THEN
    INSERT INTO daily_learning_activity (user_id, course_id, activity_date, seconds_watched, assets_viewed)
    VALUES (NEW.user_id, v_course, CURRENT_DATE, 0, 1)
    ON CONFLICT (user_id, course_id, activity_date)
    DO UPDATE SET assets_viewed = daily_learning_activity.assets_viewed + EXCLUDED.assets_viewed;
  END IF;
  RETURN NEW;
END;
$$;

CREATE OR REPLACE FUNCTION public.fn_user_points_sync_from_activities() RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE
  v_points int  := COALESCE(NEW.points, 0);
  v_day    date := (NEW.date_logged)::date;
BEGIN
  INSERT INTO user_points (user_id, period_type, period_start_date, points)
  VALUES (NEW.user_id, 'all', DATE '1970-01-01', v_points)
  ON CONFLICT (user_id, period_type, period_start_date)
  DO UPDATE SET points = user_points.points + EXCLUDED.points;
  INSERT INTO user_points (user_id, period_type, period_start_date, points)
  VALUES (NEW.user_id, 'daily', v_day, v_points)
  ON CONFLICT (user_id, period_type, period_start_date)
  DO UPDATE SET points = user_points.points + EXCLUDED.points;
  RETURN NEW;
END;
$$;

CREATE OR REPLACE FUNCTION public.mk_period_key(p_period text, p_refdate date DEFAULT CURRENT_DATE) RETURNS text LANGUAGE sql AS $$
SELECT CASE
  WHEN lower(p_period) = 'daily'   THEN 'daily:'   || to_char(p_refdate, 'YYYY-MM-DD')
  WHEN lower(p_period) = 'weekly'  THEN 'weekly:'  || to_char(date_trunc('week', p_refdate), 'IYYY-"W"IW')
  WHEN lower(p_period) = 'monthly' THEN 'monthly:' || to_char(date_trunc('month', p_refdate), 'YYYY-MM')
  WHEN lower(p_period) = 'all'     THEN 'all'
  ELSE NULL
END;
$$;

CREATE OR REPLACE FUNCTION public.recompute_kpis_for_user_course(p_user integer, p_course integer) RETURNS void LANGUAGE plpgsql AS $$
DECLARE
  v_total_dur  NUMERIC := 0;
  v_watched    NUMERIC := 0;
  v_completion NUMERIC := 0;
  v_minutes_7d NUMERIC := 0;
  v_streak     INT := 0;
  d DATE := CURRENT_DATE;
BEGIN
  SELECT COALESCE(SUM(u.duration_sec),0)::NUMERIC, COALESCE(SUM(LEAST(u.position_sec, u.duration_sec)),0)::NUMERIC
  INTO v_total_dur, v_watched
  FROM user_video_progress u JOIN course_assets a ON a.id = u.asset_id
  WHERE a.course_id = p_course AND a.kind = 'video' AND u.user_id = p_user;
  IF v_total_dur > 0 THEN v_completion := ROUND((v_watched / v_total_dur) * 100.0, 2);
  ELSE v_completion := 0; END IF;
  SELECT COALESCE(SUM(seconds_watched),0) / 60.0 INTO v_minutes_7d
  FROM daily_learning_activity WHERE user_id = p_user AND course_id = p_course AND activity_date >= CURRENT_DATE - INTERVAL '6 days';
  v_streak := 0;
  WHILE d >= CURRENT_DATE - INTERVAL '365 days' LOOP
    EXIT WHEN NOT EXISTS (SELECT 1 FROM daily_learning_activity WHERE user_id = p_user AND course_id = p_course AND activity_date = d AND seconds_watched > 0);
    v_streak := v_streak + 1;
    d := d - INTERVAL '1 day';
  END LOOP;
  INSERT INTO user_kpis (user_id, course_id, kpi_key, value_numeric, updated_at) VALUES
    (p_user, p_course, 'video_completion_pct',  v_completion,           NOW()),
    (p_user, p_course, 'engagement_minutes_7d', ROUND(v_minutes_7d,2),  NOW()),
    (p_user, p_course, 'streak_days',           v_streak,               NOW())
  ON CONFLICT (user_id, course_id, kpi_key)
  DO UPDATE SET value_numeric = EXCLUDED.value_numeric, updated_at = EXCLUDED.updated_at;
END;
$$;

CREATE OR REPLACE FUNCTION public.uvp_after_change_recompute() RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE v_course INTEGER;
BEGIN
  SELECT course_id INTO v_course FROM course_assets WHERE id = NEW.asset_id;
  PERFORM recompute_kpis_for_user_course(NEW.user_id, v_course);
  RETURN NEW;
END;
$$;
